rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    function isAdmin() {
      return request.auth != null
        && request.auth.token.email_verified == true
        && request.auth.token.email == 'wilkaaffah@gmail.com';
    }

    // like/komentar oleh user lain: hanya field likes, likedBy, comments;
    // likedBy hanya boleh berubah pada uid sendiri; komentar tidak boleh berkurang
    function engagementOnly() {
      let before = resource.data.get('likedBy', []).toSet();
      let after = request.resource.data.get('likedBy', []).toSet();
      return request.resource.data.diff(resource.data).affectedKeys().hasOnly(['likes', 'likedBy', 'comments'])
        && request.resource.data.get('likes', 0) == after.size()
        && after.difference(before).union(before.difference(after)).hasOnly([request.auth.uid])
        && request.resource.data.get('comments', []).size() >= resource.data.get('comments', []).size();
    }

    match /users/{uid} {
      allow read: if request.auth != null && (request.auth.uid == uid || isAdmin());
      allow create, update: if request.auth != null && request.auth.uid == uid;
      allow delete: if isAdmin();
    }

    match /settings/{name} {
      allow read: if true;
      allow write: if isAdmin();
    }

    match /media_items/{id} {
      // semua orang boleh melihat (login atau tidak)
      allow read: if true;

      // upload: wajib login Google, dan atas nama sendiri
      allow create: if request.auth != null
        && request.resource.data.ownerId == request.auth.uid
        && request.resource.data.title is string
        && request.resource.data.title.size() <= 200;

      // pemilik/admin boleh ubah apa saja; user lain hanya boleh like (atas nama sendiri) & menambah komentar
      allow update: if request.auth != null
        && request.resource.data.ownerId == resource.data.ownerId
        && (resource.data.ownerId == request.auth.uid || isAdmin() || engagementOnly());
      allow delete: if request.auth != null
        && (resource.data.ownerId == request.auth.uid || isAdmin());
    }

    match /profiles/{uid} {
      allow read: if true;
      allow write: if request.auth != null && request.auth.uid == uid;
    }

    match /spotlight/{id} {
      allow read: if true;
      allow write: if isAdmin();
    }

    // token notifikasi push: hanya milik sendiri (server memakai Admin SDK)
    match /push_tokens/{token} {
      allow create, update: if request.auth != null && request.resource.data.uid == request.auth.uid;
      allow read, delete: if request.auth != null && resource.data.uid == request.auth.uid;
    }
  }
}
